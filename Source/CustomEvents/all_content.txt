namespace RimTelligence.Source.CustomEvents.Execution
{
    public static class LlmEffectExecutor
    {
        public static GlobalTargetInfo ExecuteAll(LlmCustomEvent evt, Map map)
        {
            if (evt?.effects == null || !evt.effects.Any())
            {
                ModLogger.Warning("Execute: no effects to run.");
                return GlobalTargetInfo.Invalid;
            }

            ModLogger.Message($"Running {evt.effects.Count} effects...");
            GlobalTargetInfo focusTarget = GlobalTargetInfo.Invalid;

            foreach (var effect in evt.effects)
            {
                var handler = EffectHandlerRegistry.GetHandler(effect.type);
                if (handler == null)
                {
                    ModLogger.Warning($"Unknown effect type requested by LLM: {effect.type}");
                    continue;
                }

                ModLogger.Message($"Executing: {handler.EffectType} (def: {effect.defName}, count: {effect.count})");

                try
                {
                    GlobalTargetInfo result = handler.Execute(effect, map);
                    if (!focusTarget.IsValid && result.IsValid)
                    {
                        focusTarget = result;
                    }
                }
                catch (Exception e)
                {
                    ModLogger.Error($"Effect {effect.type} failed: {e.Message}\n{e.StackTrace}");
                }
            }

            return focusTarget;
        }

        public static string BuildSummary(LlmCustomEvent evt)
        {
            if (evt?.effects == null || !evt.effects.Any()) return string.Empty;

            var lines = new StringBuilder();
            lines.AppendLine("Effects:");

            foreach (var effect in evt.effects)
            {
                var handler = EffectHandlerRegistry.GetHandler(effect.type);
                string summaryLine = handler != null
                    ? handler.GetSummary(effect)
                    : $"  • Unknown Effect: {effect.type} ({effect.defName})";

                lines.AppendLine(summaryLine);
            }

            return lines.ToString().TrimEnd();
        }
    }
}

namespace RimTelligence.Source.CustomEvents.Execution
{
    public class RimTelligenceDispatcher : GameComponent
    {
        // Thread-safe queue for actions coming from the LLM
        public static readonly ConcurrentQueue<Action> ExecutionQueue = new ConcurrentQueue<Action>();

        public RimTelligenceDispatcher(Game game) { }

        public override void GameComponentTick()
        {
            base.GameComponentTick();
            // Process up to 5 actions per tick to avoid stuttering if multiple arrive
            int processed = 0;
            while (processed < 5 && ExecutionQueue.TryDequeue(out Action action))
            {
                try
                {
                    action?.Invoke();
                }
                catch (Exception ex)
                {
                    Log.Error($"[RimTelligence] Error executing queued action: {ex.Message}\n{ex.StackTrace}");
                }
                processed++;
            }
        }
    }
}

namespace RimTelligence.Source.CustomEvents.Handlers
{
    public class AlphabeaversHandler : IEffectHandler
    {
        public string EffectType => "Alphabeavers";
        public string PromptDescription => "- Alphabeavers   : count = how many spawn";

        public string AditionalPromptInfo => "";

        public GlobalTargetInfo Execute(LlmEffect effect, Map map)
        {
            PawnKindDef beaverKind = PawnKindDef.Named("Alphabeaver");
            if (beaverKind == null)
            {
                ModLogger.Warning("Alphabeavers: PawnKindDef 'Alphabeaver' not found.");
                return GlobalTargetInfo.Invalid;
            }

            int count = effect.count > 0 ? effect.count : 3;
            IntVec3 spawnCell = CellFinder.RandomEdgeCell(map);
            Pawn lastBeaver = null;

            for (int i = 0; i < count; i++)
            {
                PawnGenerationRequest request = new PawnGenerationRequest(beaverKind, null);
                Pawn beaver = PawnGenerator.GeneratePawn(request);
                GenSpawn.Spawn(beaver, CellFinder.RandomEdgeCell(map), map);
                lastBeaver = beaver;
                ModLogger.Message($"Spawned alphabeaver at {spawnCell}.");
            }

            return lastBeaver != null ? new GlobalTargetInfo(lastBeaver) : GlobalTargetInfo.Invalid;
        }

        public string GetSummary(LlmEffect effect) => $"{effect.count} alphabeaver(s) spawned";
    }
}

namespace RimTelligence.Source.CustomEvents.Handlers
{
    public class AnimalInsanityHandler : IEffectHandler
    {
        public string EffectType => "AnimalInsanity";
        public string PromptDescription => "- AnimalInsanity : defName = PawnKindDef of animal, count = how many go manhunter";

        public string AditionalPromptInfo => "select only animals from [WILDLIFE] list";

        public GlobalTargetInfo Execute(LlmEffect effect, Map map)
        {
            List<Pawn> candidates = map.mapPawns.AllPawnsSpawned
                .Where(p => p.RaceProps.Animal && !p.InMentalState)
                .ToList();

            
            candidates = candidates.Where(p => p.kindDef.defName == effect.defName).ToList();
            

            if (!candidates.Any())
            {
                ModLogger.Warning($"AnimalInsanity: no valid animals found for '{effect.defName}'.");
                return GlobalTargetInfo.Invalid;
            }

            int count = effect.count > 0 ? Math.Min(effect.count, candidates.Count) : 1;
            candidates = candidates.InRandomOrder().Take(count).ToList();

            foreach (Pawn animal in candidates)
            {
                animal.mindState.mentalStateHandler.TryStartMentalState(MentalStateDefOf.Manhunter, forceWake: true);
                ModLogger.Message($"Drove {animal.LabelShort} manhunter.");
            }

            return new GlobalTargetInfo(candidates.First());
        }

        public string GetSummary(LlmEffect effect) =>
            $"{effect.count} animal(s) {effect.defName} went manhunter";
    }
}

namespace RimTelligence.Source.CustomEvents.Handlers
{
    public class CropBlightHandler : IEffectHandler
    {
        public string EffectType => "CropBlight";

        public string PromptDescription => "- CropBlight    : defName = plant ThingDef to blight (e.g. Plant_Corn, Plant_Rice) or \"random\", value = number of plants to blight, targetPawn is unused";

        public string AditionalPromptInfo => "select defName only from harvestable_plants list.";

        public GlobalTargetInfo Execute(LlmEffect effect, Map map)
        {
            int count = Math.Max(1, (int)effect.value);

            IEnumerable<Plant> candidates = map.listerThings
                .ThingsInGroup(ThingRequestGroup.Plant)
                .OfType<Plant>()
                .Where(p => p.BlightableNow);

            if (!effect.defName.Equals("random", StringComparison.OrdinalIgnoreCase))
            {
                ThingDef cropDef = DefDatabase<ThingDef>.GetNamed(effect.defName, errorOnFail: false);
                if (cropDef != null)
                {
                    candidates = candidates.Where(p => p.def == cropDef);
                }
                else
                {
                    ModLogger.Warning($"CropBlight: Unknown plant def '{effect.defName}', falling back to any eligible crop.");
                }
            }

            List<Plant> targets = candidates.InRandomOrder().Take(count).ToList();

            if (targets.Count == 0)
            {
                ModLogger.Warning($"CropBlight: No BlightableNow crops found for '{effect.defName}'.");
                return GlobalTargetInfo.Invalid;
            }

            foreach (Plant plant in targets)
            {
                plant.CropBlighted();
            }

            string cropLabel = effect.defName.Equals("random", StringComparison.OrdinalIgnoreCase)
                ? "random crop(s)"
                : effect.defName;

            ModLogger.Message($"CropBlight: Blighted {targets.Count} {cropLabel}.");

            return new GlobalTargetInfo(targets[0].Position, map);
        }

        public string GetSummary(LlmEffect effect) =>
            $"{effect.value} {(effect.defName.Equals("random", StringComparison.OrdinalIgnoreCase) ? "random crop(s)" : effect.defName)} blighted";
    }
}

namespace RimTelligence.Source.CustomEvents.Handlers.Implementations
{
    public class LoseItemsHandler : IEffectHandler
    {
        public string EffectType => "LoseItems";

        public string PromptDescription => "- LoseItems      : defName = ThingDef (Silver, Wort, etc. ), count = amount";

        public string AditionalPromptInfo => "select only from itmes from Colony resources it can affect only items that are inside colony stockpile";

        public GlobalTargetInfo Execute(LlmEffect effect, Map map)
        {
            int leftToLose = effect.count;
            if (leftToLose <= 0 || string.IsNullOrWhiteSpace(effect.defName)) return GlobalTargetInfo.Invalid;

            foreach (Zone_Stockpile stockpile in map.zoneManager.AllZones.OfType<Zone_Stockpile>())
            {
                foreach (IntVec3 cell in stockpile.Cells)
                {
                    List<Thing> things = cell.GetThingList(map).ToList();
                    foreach (Thing thing in things)
                    {
                        if (leftToLose <= 0) return GlobalTargetInfo.Invalid;

                        if (string.Equals(thing.def.defName, effect.defName, StringComparison.OrdinalIgnoreCase))
                        {
                            int take = Math.Min(thing.stackCount, leftToLose);
                            thing.SplitOff(take).Destroy(DestroyMode.Vanish);
                            leftToLose -= take;
                        }
                    }
                }
            }
            return GlobalTargetInfo.Invalid;
        }

        public string GetSummary(LlmEffect effect) =>
            $"{effect.count} items(s) {effect.defName} lost";
        
    }
}

    internal class ResourcePodCrashHandler : IEffectHandler
    {
        public string EffectType => "ResourcePodCrash";
        public string PromptDescription => "- ResourcePodCrash      : defName = ThingDef (Silver, Wort, etc. ), count = amount";

        public string AditionalPromptInfo => $"select only from these itmes {string.Join(", ", DataList.ResourceList)} {string.Join(", ", DataList.FoodList)}";

        public GlobalTargetInfo Execute(LlmEffect effect, Map map)
        {

            ThingDef def = DefDatabase<ThingDef>.GetNamed(effect.defName, errorOnFail: false);
            if (def == null)
            {
                ModLogger.Warning($"ResourcePodCrash: ThingDef '{effect.defName}' not found.");
                return GlobalTargetInfo.Invalid;
            }

            int numOfDropPods = Rand.RangeInclusive(1, 4);

            int baseCount = effect.count / numOfDropPods;
            int remainder = effect.count % numOfDropPods;



            List<Thing> items = new List<Thing>();

            for (int i =0; i<numOfDropPods; i++)
            {
                int thisStack = baseCount + (i < remainder ? 1 : 0);
                int remaining = thisStack;

                while (remaining > 0)
                {
                    Thing item = ThingMaker.MakeThing(def);
                    int stack = Math.Min(remaining, def.stackLimit);
                    item.stackCount = stack;
                    items.Add(item);
                    remaining -= stack;
                }
            }
            
            IntVec3 dropCell = DropCellFinder.RandomDropSpot(map);

            DropPodUtility.DropThingsNear(dropCell, map, items);

            return new GlobalTargetInfo(dropCell, map);

        }

        public string GetSummary(LlmEffect effect) => $"{effect.count}x {effect.defName} dropped";
    }
}

                p.LabelShort.Equals(effect.targetPawn, StringComparison.OrdinalIgnoreCase));

            if (target == null)
            {
                ModLogger.Warning($"AddExpPoints: no spawned colonist found for '{effect.targetPawn}'.");
                return GlobalTargetInfo.Invalid;
            }

            SkillDef skillDef = DefDatabase<SkillDef>.GetNamed(effect.defName, errorOnFail: false);

            if (skillDef == null)
            {
                ModLogger.Warning($"LoseExpPoints: SkillDef '{effect.defName}' not found.");
                return GlobalTargetInfo.Invalid;
            }

            int expPoints = GetXpReward(target.skills.GetSkill(skillDef).Level, effect.value);


            target.skills.Learn(skillDef, expPoints);

            ModLogger.Message($"AddExpPoints: added {expPoints} XP to '{effect.targetPawn}' for skill '{effect.defName}'.");

            return new GlobalTargetInfo(target);
        }

        private static int GetXpToNextLevel(int level)
        {
            int[] xpTable = new int[]
            {
                1000,2000,3000,4000,5000,
                6000,7000,8000,9000,10000,
                12000,14000,16000,18000,20000,
                22000,24000,26000,28000,30000,
                32000
            };

            return xpTable[Math.Min(level, xpTable.Length - 1)];
        }

        public static int GetXpReward(int level, float value)
        {
            int xpToNext = GetXpToNextLevel(level);

            float factor;
            if (value > 80)
            {
                factor = 0.6f;
                SetXpBonusName("huge");
            }
            else if (value > 51)
            {
                factor = 0.3f;
                SetXpBonusName("medium");
            }

            else
            {
                factor = 0.15f;
                SetXpBonusName("small");
            }
            return (int)(xpToNext * factor);
        }

        public string GetSummary(LlmEffect effect) =>
            $"{effect.targetPawn} got {XpBonusName} bonus exp points for '{effect.defName}'";

        public static void SetXpBonusName(string name)
        {
            XpBonusName = name;
        }
    }

    {
        public abstract string EffectType { get; }
        public abstract string PromptDescription { get; }
        public abstract string AditionalPromptInfo { get; }
        protected abstract float MoodOffset { get; }

        public GlobalTargetInfo Execute(LlmEffect effect, Map map)
        {
            Pawn target = map.mapPawns.FreeColonistsSpawned
                .FirstOrDefault(p => string.Equals(p.LabelShort, effect.targetPawn, StringComparison.OrdinalIgnoreCase));

            if (target == null)
            {
                ModLogger.Warning($"{EffectType}: No target found for '{effect.targetPawn}'.");
                return GlobalTargetInfo.Invalid;
            }

            string uniqueKey = $"RimTelligence_{EffectType}_{Math.Abs(effect.defName?.GetHashCode() ?? 0)}_{Environment.TickCount & 0xFFFF}";
            ThoughtDef def = GetOrCreateThoughtDef(uniqueKey, effect.defName, effect.defDescription, MoodOffset);

            var memory = (ThoughtMemoryDynamic)ThoughtMaker.MakeThought(def);
            memory.customLabel = effect.defName;
            memory.customDescription = effect.defDescription;

            target.needs?.mood?.thoughts?.memories?.TryGainMemory(memory);
            ModLogger.Message($"{EffectType}: Applied '{effect.defName}' ({MoodOffset:+#;-#}) to {target.LabelShort}.");

            return new GlobalTargetInfo(target);
        }

        public abstract string GetSummary(LlmEffect effect);

        private static ThoughtDef GetOrCreateThoughtDef(string defName, string label, string description, float moodOffset)
        {
            ThoughtDef existing = DefDatabase<ThoughtDef>.GetNamed(defName, errorOnFail: false);
            if (existing != null)
                return existing;

            ThoughtDef def = new ThoughtDef
            {
                defName = defName,
                thoughtClass = typeof(ThoughtMemoryDynamic),
                durationDays = 1f,
                stackedEffectMultiplier = 1f,
                stages = new List<ThoughtStage>
                {
                    new ThoughtStage
                    {
                        label          = label ?? defName,
                        description    = description ?? string.Empty,
                        baseMoodEffect = moodOffset,
                    }
                }
            };

            DefDatabase<ThoughtDef>.Add(def);
            return def;
        }
    }

    {
        public string EffectType => "DiseasePawn";

        private string specialSummary; // semi hack it works different everywhere
        public string AditionalPromptInfo => $"select only from these disease {string.Join(", ", DataList.DiseaseList)}";

        public GlobalTargetInfo Execute(LlmEffect effect, Map map)
        {
            HediffDef hediffDef = DefDatabase<HediffDef>.GetNamed(effect.defName, errorOnFail: false);
            if (hediffDef == null)
            {
                ModLogger.Warning($"Disease: HediffDef '{effect.defName}' not found.");
                return GlobalTargetInfo.Invalid;
            }

            var availablePawns = map.mapPawns.FreeColonistsSpawned;
            List<Pawn> targets = new List<Pawn>();

            Pawn named = availablePawns.FirstOrDefault(p => p.LabelShort.Equals(effect.targetPawn, StringComparison.OrdinalIgnoreCase));
            if (named != null) targets.Add(named);
            

            if (!targets.Any())
            {
                ModLogger.Warning($"Disease: no spawned targets found for '{effect.targetPawn}'.");
                return GlobalTargetInfo.Invalid;
            }

            foreach (Pawn pawn in targets)
            {
                Hediff existingDisease = pawn.health.hediffSet.GetFirstHediffOfDef(hediffDef);
                
                if (existingDisease != null)
                {
                    float severity = Rand.Range(0.02f, 0.1f);
                    specialSummary = $"{pawn.LabelShort} disease {hediffDef.defName} severity increased by {severity}";
                    existingDisease.Severity += severity;

                    pawn.health.Notify_HediffChanged(existingDisease);
                }
                else
                {
                    Hediff newDisease = HediffMaker.MakeHediff(hediffDef, pawn);
                    newDisease.Severity = 0.01f;
                    specialSummary = $"{pawn.LabelShort} got infected with {hediffDef.defName}";
                    pawn.health.AddHediff(newDisease);

                }

                ModLogger.Message(specialSummary);
            }

            return new GlobalTargetInfo(targets.First());
        }

        public string GetSummary(LlmEffect effect) =>
            specialSummary;
    }

namespace RimTelligence.Source.CustomEvents
{
    public static class LlmEffectValidator
    {
        // Types where an empty defName makes execution meaningless or will throw.
        private static readonly HashSet<string> _requiresDefName = new HashSet<string>(
            System.StringComparer.OrdinalIgnoreCase)
        {
            "AnimalInsanity",
            "SelfTame",
            "ChangeWeather",
            "CropBlight",
            "StartExtremeWeatherEvent",
            "LoseItems",
            "PawnFoundItem",
            "ResourcePodCrash",
            "SpawnItemsAtRandomLocation",
            "AddExpPoints",
            "MoodDebuff",
            "MoodBuff",
            "DiseasePawn",
            "DamagePawn",
            "LoseExpPoints",
            "MiracleHealChronicCondition",
            "MiracleHealDisease",
            "MiracleHealMissingPart"
        };

        public static List<LlmEffect> StripInvalid(List<LlmEffect> effects)
        {
            if (effects == null) return new List<LlmEffect>();

            var valid = new List<LlmEffect>();

            foreach (var e in effects)
            {
                string reason = GetRejectionReason(e);
                if (reason != null)
                {
                    ModLogger.Warning($"[Validator] Dropping effect type='{e.type}' defName='{e.defName}': {reason}");
                    continue;
                }
                valid.Add(e);
            }

            return valid;
        }

        private static string GetRejectionReason(LlmEffect e)
        {
            if (string.IsNullOrWhiteSpace(e.type))
                return "type is blank";

            // Unknown type — handler won't exist, executor already warns but we drop early
            if (!Handlers.EffectHandlerRegistry.GetAllHandlers()
                    .Any(h => string.Equals(h.EffectType, e.type, System.StringComparison.OrdinalIgnoreCase)))
                return $"unknown effect type '{e.type}'";

            // Required defName missing
            if (_requiresDefName.Contains(e.type) && string.IsNullOrWhiteSpace(e.defName))
                return $"type '{e.type}' requires a defName but got empty string";

            // Pure filler: nothing meaningful was set
            bool noCount = e.count <= 0;
            bool noValue = e.value <= 0;
            bool noTarget = string.IsNullOrWhiteSpace(e.targetPawn);
            bool noDefName = string.IsNullOrWhiteSpace(e.defName);

            if (noCount && noValue && noTarget && noDefName)
                return "all fields are zero/empty — pure filler";

            return null;
        }
    }
}

            // ── Examples ──────────────────────────────────────────────────────
            sb.AppendLine("EXAMPLES:");
            sb.AppendLine();
            sb.AppendLine("// World event example:");
            sb.AppendLine("{");
            sb.AppendLine("  \"effects\": [");
            sb.AppendLine("    { \"type\": \"CropBlight\", \"defName\": \"Plant_Rice\", \"count\": 0, \"value\": 12, \"targetPawn\": \"\" }");
            sb.AppendLine("  ]");
            sb.AppendLine("  \"letterTitle\": \"Blight in the North Fields\",");
            sb.AppendLine("  \"letterText\": \"The rice crop in the north fields has turned black overnight. No one saw it happen. The colonists noticed at dawn — half the harvest, gone.\",");
            sb.AppendLine("}");
            sb.AppendLine();
            sb.AppendLine("// Pawn event example (two matching effects):");
            sb.AppendLine("{");
            sb.AppendLine("  \"effects\": [");
            sb.AppendLine("    { \"type\": \"PawnFoundItem\",  \"defName\": \"MedicineIndustrial\", \"count\": 10, \"value\": 0, \"targetPawn\": \"Vasquez\" },");
            sb.AppendLine("    { \"type\": \"AnimalInsanity\", \"defName\": \"Muffalo\",            \"count\": 1,  \"value\": 0, \"targetPawn\": \"\" }");
            sb.AppendLine("  ]");
            sb.AppendLine("  \"letterTitle\": \"Cache and Consequences\",");
            sb.AppendLine("  \"letterText\": \"Vasquez was hauling slag when a hidden supply crate cracked open under her boot, spilling industrial medicine across the floor. The commotion spooked a nearby muffalo, which promptly decided Vasquez had started it.\",");
            sb.AppendLine("}");
            sb.AppendLine();

            // ── Effect types, split into world vs pawn ─────────────────────────
            // Splitting the list makes it visually obvious which bucket to draw
            // from when the focus steering says "prefer world effects".
            var allHandlers = EffectHandlerRegistry.GetAllHandlers().ToList();
            var worldHandlers = allHandlers.Where(h => _worldEffectTypes.Contains(h.EffectType)).ToList();
            var pawnHandlers = allHandlers.Where(h => !_worldEffectTypes.Contains(h.EffectType)).ToList();

            sb.AppendLine("WORLD EFFECTS (colony/environment — no specific pawn required):");
            foreach (var h in worldHandlers)
            {
                string line = $"  {h.PromptDescription}";
                if (!string.IsNullOrWhiteSpace(h.AditionalPromptInfo))
                    line += $"  — Allowed values: {h.AditionalPromptInfo}";
                sb.AppendLine(line);
            }
            sb.AppendLine();

            sb.AppendLine("PAWN EFFECTS (target a specific colonist):");
            foreach (var h in pawnHandlers)
            {
                string line = $"  {h.PromptDescription}";
                if (!string.IsNullOrWhiteSpace(h.AditionalPromptInfo))
                    line += $"  — Allowed values: {h.AditionalPromptInfo}";
                sb.AppendLine(line);
            }
            sb.AppendLine();

            // ── Colony state ───────────────────────────────────────────────────
            sb.AppendLine("COLONY STATE (your story seed):");
            sb.AppendLine(snapshotNarrative);
            sb.AppendLine();

            sb.AppendLine("Return ONLY the JSON object.");

            return sb.ToString();
        }
    }

   public static class SnapshotNarrator
    {
        private static readonly Random _rng = new Random();

        private const int MaxResourceEntries = 10;
        private const int SpotlightCount = 2;   
        private const int BriefCount = 999;

        public static string Narrate(ColonySnapshot s)
        {
            var sb = new StringBuilder();

            sb.Append($"Colony \"{s.colonyName}\", day {s.day}");
            sb.Append($" — {s.season?.ToLower() ?? "unknown season"}, {s.biome?.ToLower() ?? "hostile biome"}.");
            sb.Append($" Outside: {DescribeTemperature(s.temperature)}, {s.currentWeather?.ToLower() ?? "unclear weather"}.");
            sb.AppendLine();


            if (s.colonists != null && s.colonists.Count > 0)
            {
                var names = s.colonists.Select(p => p.name).Where(n => !string.IsNullOrWhiteSpace(n)).ToList();
                sb.AppendLine($"Colonists ({s.colonistCount} total, exact names for targetPawn): {string.Join(", ", names)}.");
                sb.AppendLine();

                var ordered = s.colonists.OrderByDescending(InterestScore).ToList();
                var spotlight = ordered.Take(SpotlightCount).ToList();
                var brief = ordered.Skip(SpotlightCount).ToList();

                sb.AppendLine("Key colonists (full detail):");
                foreach (var p in spotlight)
                    sb.AppendLine("  " + DescribePawnFull(p));

                if (brief.Count > 0)
                {
                    sb.AppendLine("Other colonists (brief):");
                    foreach (var p in brief)
                        sb.AppendLine("  " + DescribePawnBrief(p));
                }
                sb.AppendLine();
            }

            sb.AppendLine(DescribeFood(s.totalNutrition, s.colonistCount));
            sb.AppendLine(DescribeWealth(s.wealthTotal));
            sb.AppendLine(DescribeRaidPressure(s.daysSinceLastRaid));

            if (s.activeConditions != null && s.activeConditions.Count > 0)
                sb.AppendLine($"Active conditions: {string.Join(", ", s.activeConditions)}.");

            if (s.wildAnimals != null && s.wildAnimals.Count > 0)
            {
                var sample = s.wildAnimals
                    .OrderByDescending(kv => kv.Value)
                    .Take(6)
                    .Select(kv => $"{kv.Value}x {kv.Key}")
                    .ToList();
                sb.AppendLine($"Wildlife nearby: {string.Join(", ", sample)}.");
            }

            if (s.cultivatedPlants != null && s.cultivatedPlants.Count > 0)
            {
                var crops = s.cultivatedPlants
                    .Select(kv => $"{kv.Value}x {kv.Key}")
                    .ToList();
                sb.AppendLine($"Crops: {string.Join(", ", crops)}.");
            }

            if (s.resources?.allResources != null && s.resources.allResources.Count > 0)
            {
                var topResources = s.resources.allResources
                    .OrderByDescending(kv => kv.Value)
                    .Take(MaxResourceEntries)
                    .Select(kv => $"{kv.Key}: {kv.Value}");
                string suffix = s.resources.allResources.Count > MaxResourceEntries
                    ? $" (+ {s.resources.allResources.Count - MaxResourceEntries} more)"
                    : "";
                sb.AppendLine($"Resources: {string.Join(", ", topResources)}{suffix}.");
            }

            if (s.recentEventTitles != null && s.recentEventTitles.Count > 0)
                sb.AppendLine($"[RECENT EVENTS — do not repeat]: {string.Join("; ", s.recentEventTitles)}.");

            return sb.ToString().Trim();
        }

        private static float InterestScore(PawnSnapshot p)
        {
            float score = 0f;
            if (p.health?.is_downed == true) score += 10f;
            if (p.health?.diseases?.Count > 0) score += 5f;
            if (p.health?.injuries?.Count > 0) score += 3f;
            if (p.health?.missing_body_parts?.Count > 0) score += 4f;
            if (p.mood < 0.3f) score += 4f;
            else if (p.mood < 0.5f) score += 2f;
            if (p.traits?.Count > 0) score += 1f;
            return score;
        }

        private static string DescribePawnFull(PawnSnapshot p)
        {
            var sb = new StringBuilder();
            sb.Append($"{p.name} (age {p.age})");

            if (!string.IsNullOrWhiteSpace(p.current_action))
                sb.Append($" [ACTION: {p.current_action.ToLower()}]");

            if (p.queued_actions?.Count > 0)
                sb.Append($" [QUEUE: {string.Join(" → ", p.queued_actions).ToLower()}]");

            sb.Append($" [MOOD: {DescribeMood(p.mood)}]");

            if (p.health != null)
            {
                sb.Append($" [HEALTH: {p.health.overall_health:P0}");
                if (p.health.is_downed) sb.Append(", DOWNED");
                if (p.health.pain_level > 0.3f) sb.Append($", pain {p.health.pain_level:P0}");
                if (p.health.injuries?.Count > 0) sb.Append($", injuries: {string.Join(", ", p.health.injuries)}");
                if (p.health.diseases?.Count > 0) sb.Append($", diseases: {string.Join(", ", p.health.diseases)}");
                if (p.health.missing_body_parts?.Count > 0) sb.Append($", missing: {string.Join(", ", p.health.missing_body_parts)}");
                if (p.health.implants?.Count > 0) sb.Append($", implants: {string.Join(", ", p.health.implants)}");
                if (p.health.chronic_conditions?.Count > 0) sb.Append($", chronic: {string.Join(", ", p.health.chronic_conditions)}");
                sb.Append("]");
            }

            if (p.traits?.Count > 0)
                sb.Append($" [TRAITS: {string.Join(", ", p.traits)}]");

            // Background only in full detail
            if (!string.IsNullOrWhiteSpace(p.childhood) || !string.IsNullOrWhiteSpace(p.adulthood))
                sb.Append($" [PAST: {p.childhood}/{p.adulthood}]");

            if (p.skills?.Count > 0)
                sb.Append($" [SKILLS: {string.Join(", ", p.skills)}]");

            return sb.ToString();
        }

        private static string DescribePawnBrief(PawnSnapshot p)
        {
            var sb = new StringBuilder();
            sb.Append($"{p.name}");

            if (!string.IsNullOrWhiteSpace(p.current_action))
                sb.Append($" [ACTION: {p.current_action.ToLower()}]");

            sb.Append($" [MOOD: {DescribeMood(p.mood)}]");

            if (p.health?.is_downed == true) sb.Append(" [DOWNED]");
            if (p.health?.diseases?.Count > 0) sb.Append($" [SICK: {string.Join(", ", p.health.diseases)}]");
            if (p.traits?.Count > 0) sb.Append($" [TRAITS: {string.Join(", ", p.traits)}]");

            return sb.ToString();
        }


        private static string DescribeMood(float mood)
        {
            if (mood < 0.25f) return $"{mood:P0} (breaking)";
            if (mood < 0.45f) return $"{mood:P0} (miserable)";
            if (mood < 0.65f) return $"{mood:P0} (stable)";
            if (mood < 0.80f) return $"{mood:P0} (good)";
            return $"{mood:P0} (euphoric)";
        }

        private static string DescribeFood(float nutrition, int colonistCount)
        {
            if (colonistCount <= 0) return "Food situation unknown.";
            float daysLeft = nutrition / Math.Max(colonistCount * 1.6f, 1f);
            if (daysLeft < 2) return "The colony is starving. Meals are rationed to crumbs.";
            if (daysLeft < 5) return "Food is dangerously low — hunger is a constant companion.";
            if (daysLeft < 12) return "Provisions are modest; nobody starving, nobody comfortable.";
            if (daysLeft < 25) return "Food stocks are adequate.";
            return "The pantry is embarrassingly full. Abundance breeds complacency.";
        }

        private static string DescribeWealth(float wealth)
        {
            if (wealth < 15000) return "The colony is dirt poor — scavenged furniture and desperate hope.";
            if (wealth < 40000) return "Scraping by: enough to survive, little to spare.";
            if (wealth < 100000) return "A modest, functional colony — not impressive, but standing.";
            if (wealth < 250000) return "The colony has accumulated real wealth. Comfortable, almost.";
            return "Grotesquely wealthy by rimworld standards. A tempting target.";
        }

        private static string DescribeRaidPressure(int daysSince)
        {
            if (daysSince < 0) return "Raiders have not come yet — but they will.";
            if (daysSince == 0) return "Raiders struck today. The blood is still fresh.";
            if (daysSince < 4) return "The last raid was only days ago. Wounds haven't closed.";
            if (daysSince < 12) return "It has been quiet since the last raid. Suspiciously quiet.";
            if (daysSince < 25) return "Weeks without violence. The colonists are getting restless.";
            return "It has been a very long time since the last raid. Something must be coming.";
        }

        private static string DescribeTemperature(float celsius)
        {
            if (celsius < -30) return "lethally cold";
            if (celsius < -10) return "bitterly freezing";
            if (celsius < 5) return "cold";
            if (celsius < 18) return "cool";
            if (celsius < 28) return "pleasant";
            if (celsius < 38) return "sweltering";
            return "dangerously hot";
        }
    }

    public class IncidentWorker_ProceduralEvent : IncidentWorker
    {
        protected override bool TryExecuteWorker(IncidentParms parms)
        {
            if (!(parms.target is Map map))
            {
                ModLogger.Error("TryExecuteWorker: parms.target is not a Map.");
                return false;
            }

            ModLogger.Message("ProceduralEvent triggered, capturing snapshot...");
            FireLlmEventAsync(map);
            return true;
        }

        private async void FireLlmEventAsync(Map map)
        {
            // 1. Data Collection
            ColonySnapshot snapshot = ColonySnapshotCapturer.Capture(map);
            string snapshotJson = JsonConvert.SerializeObject(snapshot, Formatting.Indented);

            string snapshotNarrative = SnapshotNarrator.Narrate(snapshot);

            // 2. Prompt Generation
            string prompt = PromptGenerator.BuildSystemPrompt(snapshotNarrative);
            ModLogger.Message($"Prompt generated: {prompt}");
            ModLogger.Message($"Sending prompt to LLM ({prompt.Length} chars)");

            // 3. External API Call
            BasicCom com = new BasicCom();
            string response;
            try
            {
                response = await com.GetLlmResponse(prompt);
            }
            catch (Exception e)
            {
                ModLogger.Error($"LLM call failed: {e.Message}");
                return;
            }

            if (string.IsNullOrWhiteSpace(response))
            {
                ModLogger.Error("LLM returned empty response.");
                return;
            }
            ModLogger.Message($"LLM response received: {response}");

            // 4. Data Parsing & Sanitization
            string cleanedJson = response.Replace("```json", "").Replace("```", "").Trim();
            LlmCustomEvent evt;

            try
            {
                evt = JsonConvert.DeserializeObject<LlmCustomEvent>(cleanedJson);
            }
            catch (Exception e)
            {
                ModLogger.Error($"JSON parse failed: {e.Message}\nResponse was:\n{cleanedJson}");
                return;
            }

            if (evt == null)
            {
                ModLogger.Error($"JSON deserialized to null.\nResponse was:\n{cleanedJson}");
                return;
            }

            // 4b. Strip effects the LLM emitted incorrectly (missing defName, all-zero, unknown type)
            evt.effects = LlmEffectValidator.StripInvalid(evt.effects);



            // 5. Game Thread Execution
            RimTelligenceDispatcher.ExecutionQueue.Enqueue(() =>
            {
                GlobalTargetInfo target = LlmEffectExecutor.ExecuteAll(evt, map);

                string summary = LlmEffectExecutor.BuildSummary(evt);
                string fullText = $"{evt.letterText ?? "Something happened."}\n\n{summary}";

                Find.LetterStack.ReceiveLetter(
                    evt.letterTitle ?? "Strange Event",
                    fullText,
                    LetterDefOf.ThreatSmall,
                    target.IsValid ? new LookTargets(target) : null
                );
            });
        }
    }

    public class Dialog_EffectTest : Window
    {
        private readonly Map map;
        private readonly LlmEffect effect;

        private string defNameBuf;
        private string defDescriptionBuf;
        private string pawnBuf;
        private string countBuf;
        private string valueBuf;

        private Vector2 pawnScroll;
        private Vector2 defScroll;

        private const float ButtonHeight = 30f;
        private const float ScrollSectionHeight = 150f;

        public override Vector2 InitialSize => new Vector2(480f, 620f);

        public Dialog_EffectTest(LlmEffect effect, Map map)
        {
            this.effect = effect;
            this.map = map;

            defNameBuf = effect.defName;
            defDescriptionBuf = effect.defDescription;
            pawnBuf = effect.targetPawn;
            countBuf = effect.count.ToString();
            valueBuf = effect.value.ToString();

            doCloseButton = true;
            closeOnClickedOutside = true;
        }

        public override void DoWindowContents(Rect inRect)
        {
            Listing_Standard ui = new Listing_Standard();
            ui.Begin(inRect);

            ui.Label($"Handler: {effect.type}");
            ui.GapLine();

            var handler = EffectHandlerRegistry.GetHandler(effect.type);
            if (handler != null)
            {
                ui.Label($"Hint: {handler.PromptDescription}");
                ui.Gap();
            }

            defNameBuf = ui.TextEntryLabeled("defName:    ", defNameBuf);
            defDescriptionBuf = ui.TextEntryLabeled("defDescription: ", defDescriptionBuf);
            pawnBuf = ui.TextEntryLabeled("targetPawn: ", pawnBuf);
            ui.TextFieldNumericLabeled("count: ", ref effect.count, ref countBuf, 0f, 10000f);
            ui.TextFieldNumericLabeled("value: ", ref effect.value, ref valueBuf, 0f, 10000f);

            ui.GapLine();

            // ── pawn quick-fill ──────────────────────────────────────────────
            ui.Label("Quick-fill pawn:");
            DrawScrollableButtons(
                ui, inRect,
                map.mapPawns.FreeColonists.Select(p => p.Name.ToStringShort).ToList(),
                ref pawnScroll,
                val => pawnBuf = val
            );

            // ── def quick-fill ───────────────────────────────────────────────
            List<string> defList = GetDefList(effect.type);
            if (defList != null)
            {
                ui.GapLine();
                ui.Label($"Quick-fill ({effect.type}):");
                DrawScrollableButtons(
                    ui, inRect,
                    defList,
                    ref defScroll,
                    val => defNameBuf = val
                );
            }

            ui.Gap();

            if (ui.ButtonText("Fire Effect"))
                FireEffect();

            ui.End();
        }

        private void DrawScrollableButtons(
            Listing_Standard ui,
            Rect inRect,
            List<string> items,
            ref Vector2 scroll,
            System.Action<string> onSelect)
        {
            if (items == null || items.Count == 0)
            {
                ui.Label("  (none)");
                return;
            }

            float totalHeight = items.Count * ButtonHeight;
            float viewHeight = Mathf.Min(totalHeight, ScrollSectionHeight);
            float innerWidth = inRect.width - 20f;

            Rect outerRect = ui.GetRect(viewHeight);
            Rect innerRect = new Rect(0f, 0f, innerWidth, totalHeight);

            Widgets.BeginScrollView(outerRect, ref scroll, innerRect);

            float y = 0f;
            foreach (string item in items)
            {
                Rect btnRect = new Rect(0f, y, innerWidth * 0.6f, ButtonHeight - 2f);
                if (Widgets.ButtonText(btnRect, item))
                    onSelect(item);
                y += ButtonHeight;
            }

            Widgets.EndScrollView();
            ui.Gap(4f);
        }

        private static List<string> GetDefList(string effectType)
        {
            switch (effectType)
            {
                case "AnimalInsanity": return DataList.AnimalList;
                case "DiseasePawn": return DataList.DiseaseList;
                case "MiracleHeal": return DataList.DiseaseList;
                case "PawnFoundItem": return DataList.ResourceList;
                case "Damage": return DataList.DamageTypeList;
                case "ChangeWeather": return DataList.AllWeathersList;
                case "GameCondition": return DataList.GameConditionList;
                case "AddExpPoints": return DataList.AllSkillsList;
                case "LoseExpPoints": return DataList.AllSkillsList;
                default: return null;
            }
        }

        private void FireEffect()
        {
            effect.defName = defNameBuf;
            effect.defDescription = defDescriptionBuf;
            effect.targetPawn = pawnBuf;

            LlmCustomEvent evt = new LlmCustomEvent
            {
                letterTitle = $"Test: {effect.type}",
                letterText = $"Debug fire of {effect.type}.",
                effects = new List<LlmEffect> { effect }
            };

            string testJson = JsonConvert.SerializeObject(evt);
            Log.Message($"[RimTelligence] Firing: {testJson}");

            IncidentWorker_Test worker = new IncidentWorker_Test();
            worker.FireLlmEventAsync(map, testJson);

            Close();
        }
    }
